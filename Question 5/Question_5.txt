1. Fetch Data (HTTP GET Request with Promise):
------------------------------------------------
function fetchData(url) {
  return fetch(url)
    .then(response => {
      if (!response.ok) throw new Error('Network error');
      return response.json();
    });
}

2. Recursively Compute Exponent of a Number:
------------------------------------------------
function power(base, exponent) {
  if (exponent === 0) return 1;
  return base * power(base, exponent - 1);
}

3. Find Second Lowest and Second Highest Numbers in Array:
------------------------------------------------
function secondStats(arr) {
  const uniqueSorted = [...new Set(arr)].sort((a, b) => a - b);
  if (uniqueSorted.length < 2) return [];
  return [uniqueSorted[1], uniqueSorted[uniqueSorted.length - 2]];
}

4. Determine if a Year is a Leap Year:
------------------------------------------------
function isLeapYear(year) {
  return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;
}

5. Bank Account Class with Deposit & Withdraw:
------------------------------------------------
class BankAccount {
  constructor(accountNumber, balance = 0) {
    this.accountNumber = accountNumber;
    this.balance = balance;
  }

  deposit(amount) {
    this.balance += amount;
  }

  withdraw(amount) {
    if (amount <= this.balance) {
      this.balance -= amount;
    }
  }
}

const account = new BankAccount('123');
account.deposit(100);
account.withdraw(50);

7. Binary Search Implementation:
------------------------------------------------
function binarySearch(arr, target) {
  let left = 0, right = arr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    arr[mid] < target ? left = mid + 1 : right = mid - 1;
  }

  return -1;
}
8. Convert Singly Linked List to Array:
------------------------------------------------
function linkedListToArray(head) {
  const result = [];
  let current = head;

  while (current) {
    result.push(current.value);
    current = current.next;
  }

  return result;
}
